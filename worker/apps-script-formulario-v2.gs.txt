var SHARED_SECRET = "3-estoesparaquevitarenazcabajolasreglasduras";

var COLS = {
  timestamp: 0, modo: 1, nombre_nino: 2, edad: 3, que_deseas_lograr: 4,
  preocupacion_principal: 5, origen: 6, sueno_y_movimiento: 7,
  senales_observables: 8, diagnosticos_y_atencion_medica: 9,
  cronologia_del_dia: 10, finde_y_gustos: 11, mente_y_pantallas: 12,
  emociones: 13, estado: 14, notas_peter: 15,
  id_plan: 16, plan_json: 17, decision: 18, borrador_texto: 19,
  nombre_padre: 20, email_padre: 21,
};

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);

    if (!body.secret || body.secret !== SHARED_SECRET) {
      return respond({ ok: false, error: "secret inválido" }, 401);
    }

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];

    if (body.action === "guardar_borrador") {
      return guardarBorrador(sheet, body);
    }

    if (body.action === "notificar_escalamiento") {
      return notificarEscalamiento(body);
    }

    // Comportamiento original: agregar fila nueva capturada por Vita.
    if (!body.dx) {
      return respond({ ok: false, error: "dx faltante" }, 400);
    }
    var dx = body.dx;
    sheet.appendRow([
      body.ts || new Date().toISOString(),
      body.mode || "",
      dx.nombre_nino || "", dx.edad || "", dx.que_deseas_lograr || "",
      dx.preocupacion_principal || "", dx.origen || "", dx.sueno_y_movimiento || "",
      dx.senales_observables || "", dx.diagnosticos_y_atencion_medica || "",
      dx.cronologia_del_dia || "", dx.finde_y_gustos || "", dx.mente_y_pantallas || "",
      dx.emociones || "",
      "pendiente", "",       // estado, notas_peter
      "", "", "", "",        // id_plan, plan_json, decision, borrador_texto
      dx.nombre_padre || "", dx.email_padre || ""
    ]);
    var lastRow = sheet.getLastRow();
    SpreadsheetApp.flush();
    return respond({ ok: true, lastRow: lastRow });
  } catch (err) {
    return respond({ ok: false, error: String(err) }, 500);
  }
}

function guardarBorrador(sheet, body) {
  var ts = body.ts;
  if (!ts) return respond({ ok: false, error: "ts faltante" }, 400);

  var data = sheet.getDataRange().getValues();
  for (var r = 1; r < data.length; r++) {
    if (String(data[r][COLS.timestamp]) === String(ts)) {
      var rowNum = r + 1;
      var idPlan = Utilities.getUuid().replace(/-/g, "").slice(0, 16);
      sheet.getRange(rowNum, COLS.id_plan + 1).setValue(idPlan);
      sheet.getRange(rowNum, COLS.plan_json + 1).setValue(JSON.stringify(body.plan || {}));
      sheet.getRange(rowNum, COLS.decision + 1).setValue("pendiente_aprobacion");
      sheet.getRange(rowNum, COLS.borrador_texto + 1).setValue(body.mensaje || "");
      sheet.getRange(rowNum, COLS.estado + 1).setValue("borrador_listo");
      SpreadsheetApp.flush();
      return respond({ ok: true, idPlan: idPlan, row: rowNum });
    }
  }
  return respond({ ok: false, error: "no se encontró la fila con ese ts" }, 404);
}

// Chat de seguimiento (plan/index.html) — cuando el bot detecta algo que hay
// que escalar (urgencia, duda sobre el plan, pide hablar con una persona),
// esto manda el aviso real por correo. Sin esto, el chat podría "decir" que
// lo va a escalar sin que nadie del equipo lo vea de verdad.
function notificarEscalamiento(body) {
  try {
    MailApp.sendEmail({
      to: "leonardo@ledelab.group",
      subject: "🚨 Vita escaló algo — " + (body.motivo || "sin especificar") + (body.nombreNino ? " — " + body.nombreNino : ""),
      body:
        "Motivo: " + (body.motivo || "sin especificar") + "\n" +
        "Niño/a: " + (body.nombreNino || "(sin dato)") + "\n" +
        "id_plan: " + (body.idPlan || "(sin dato)") + "\n\n" +
        "Mensaje del padre/madre:\n" + (body.mensajePadre || "(sin mensaje)"),
    });
    return respond({ ok: true });
  } catch (err) {
    return respond({ ok: false, error: String(err) }, 500);
  }
}

function doGet(e) {
  try {
    var id = e.parameter.id;
    if (!id) return respond({ ok: false, error: "id faltante" }, 400);

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    var data = sheet.getDataRange().getValues();

    for (var r = 1; r < data.length; r++) {
      if (String(data[r][COLS.id_plan]) === String(id)) {
        var decision = data[r][COLS.decision];
        if (decision !== "aprobado") {
          return respond({ ok: false, error: "aún no aprobado" }, 403);
        }
        var plan = {};
        try { plan = JSON.parse(data[r][COLS.plan_json] || "{}"); } catch (ex) {}
        return respond({
          ok: true,
          nombreNino: data[r][COLS.nombre_nino],
          mensaje: data[r][COLS.borrador_texto],
          plan: plan,
        });
      }
    }
    return respond({ ok: false, error: "no encontrado" }, 404);
  } catch (err) {
    return respond({ ok: false, error: String(err) }, 500);
  }
}

function respond(obj, status) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ============================================================
// TRIGGER DE APROBACIÓN — avisa por correo apenas alguien pone "aprobado"
// en la columna "decision". OJO: esto NO corre solo con pegar el código —
// es un trigger INSTALABLE (los simple triggers como onEdit no pueden
// mandar correo). Instálalo así, una sola vez:
//   1. En el editor de Apps Script, ícono del reloj (⏰, "Activadores") a
//      la izquierda.
//   2. + Agregar activador (esquina inferior derecha).
//   3. Función: onAprobado · Fuente del evento: Desde la hoja de cálculo ·
//      Tipo de evento: Al editar · Guardar.
//   4. Te va a pedir autorizar permisos (para poder mandar el correo) —
//      autorízalo, es tu propio script.
// ============================================================
function onAprobado(e) {
  try {
    if (!e || !e.range) return;
    if (e.range.getColumn() !== COLS.decision + 1) return; // solo nos importa "decision"
    if (String(e.value || "").trim() !== "aprobado") return;

    var sheet = e.range.getSheet();
    var row = e.range.getRow();
    var rowValues = sheet.getRange(row, 1, 1, sheet.getLastColumn()).getValues()[0];

    var nombreNino = rowValues[COLS.nombre_nino] || "";
    var nombrePadre = rowValues[COLS.nombre_padre] || "";
    var emailPadre = String(rowValues[COLS.email_padre] || "").trim();
    var idPlan = rowValues[COLS.id_plan] || "";

    if (!idPlan) {
      MailApp.sendEmail({
        to: "leonardo@ledelab.group",
        subject: "⚠️ Aprobaste un plan sin borrador — " + (nombreNino || "sin nombre"),
        body: "Pusiste 'aprobado' en la fila de " + (nombreNino || "(sin nombre)") + " pero todavía no hay un borrador generado (falta el id_plan). Genera el borrador primero.",
      });
      return;
    }

    var planUrl = "https://be360.app/plan/?id=" + idPlan;

    if (!emailPadre) {
      // Sin correo no hay a quién enviarle — avisa al equipo para que lo
      // resuelva a mano en vez de fallar en silencio.
      MailApp.sendEmail({
        to: "leonardo@ledelab.group",
        subject: "⚠️ Falta el correo del padre — " + (nombreNino || "sin nombre"),
        body: "Se aprobó el plan de " + (nombreNino || "(sin nombre)") + " pero la columna email_padre está vacía, así que no se pudo enviar automáticamente.\n\nLink del plan (envíalo tú manualmente): " + planUrl,
      });
      return;
    }

    // Envío automático y directo al padre/madre — ya no requiere que alguien
    // del equipo haga clic para mandarlo. Último paso manual del ciclo que
    // quedaba (antes era "un clic para WhatsApp"); con esto se cierra.
    MailApp.sendEmail({
      to: emailPadre,
      name: "Bienestar 360",
      subject: "El plan de " + (nombreNino || "tu hijo/a") + " ya está listo",
      body:
        "Hola" + (nombrePadre ? " " + nombrePadre : "") + ",\n\n" +
        "Ya está listo el plan de " + (nombreNino || "tu hijo/a") + ". Puedes verlo aquí:\n" +
        planUrl + "\n\n" +
        "— El equipo de Bienestar 360",
    });

    // Copia de cortesía al equipo — solo para tener registro de que salió.
    MailApp.sendEmail({
      to: "leonardo@ledelab.group",
      subject: "✅ Plan enviado automáticamente — " + (nombreNino || "sin nombre"),
      body: "Se envió el plan de " + (nombreNino || "(sin nombre)") + " directo a " + emailPadre + ".\n\nLink: " + planUrl,
    });
  } catch (err) {
    // No relanzar: un error aquí no debe romper la edición del Sheet.
  }
}
