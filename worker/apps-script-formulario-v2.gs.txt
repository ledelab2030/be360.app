function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];

    // Acciones del panel de Peter — su propio secreto, nunca el del Worker.
    if (body.action === "listar_pendientes") return conPanelSecret(body, function () { return listarPendientes(sheet); });
    if (body.action === "listar_historial") return conPanelSecret(body, function () { return listarHistorial(sheet); });
    if (body.action === "detalle_plan") return conPanelSecret(body, function () { return detallePlan(sheet, body); });
    if (body.action === "guardar_edicion_plan") return conPanelSecret(body, function () { return guardarEdicionPlan(sheet, body); });
    if (body.action === "aprobar_plan") return conPanelSecret(body, function () { return aprobarPlan(sheet, body); });
    if (body.action === "descartar_pendiente") return conPanelSecret(body, function () { return descartarPendiente(sheet, body); });
    if (body.action === "eliminar_novedad") return conPanelSecret(body, function () { return eliminarNovedad(sheet, body); });
    // NUEVO (27 oct 2026, pedido de Leonardo — sección "Equipo" de
    // equipo/index.html): lista de solo lectura de quién tiene acceso hoy.
    // No recibe "sheet" — vive en su propia pestaña, no en la de casos.
    if (body.action === "listar_equipo") return conPanelSecret(body, function () { return listarEquipo(); });
    // NUEVO (28 oct 2026, pedido de Leonardo): borra el chat completo de un
    // caso — limpieza de mensajes de prueba, ver limpiarChat().
    if (body.action === "limpiar_chat") return conPanelSecret(body, function () { return limpiarChat(sheet, body); });
    // NUEVO (28 oct 2026, pedido de Leonardo — "evitar fricción en las
    // validaciones que siguen": login real por persona en equipo/index.html,
    // en vez de un PIN único compartido, SIN montar un sistema de auth
    // nuevo). Prueba un PIN candidato contra la columna PIN de la pestaña
    // "Equipo" (o el PIN maestro) y dice DE QUIÉN es — a propósito NO lleva
    // conPanelSecret: probar la credencial es justo lo que hace esta acción,
    // pedirla de antemano sería circular. Ver pinEsValido() más abajo, que
    // ahora es también lo que usa conPanelSecret() para las demás acciones.
    if (body.action === "login_equipo") return loginEquipo(body);

    // Verificación de código de colegio — viene de verificar/index.html (una
    // página pública, el padre la abre desde el correo). NO usa SHARED_SECRET
    // ni PANEL_SECRET: la protección es la misma que ya usa /plan — hace
    // falta conocer el idPlan (token aleatorio de 16 caracteres), no
    // enumerable. El código en sí es la credencial real.
    if (body.action === "verificar_codigo") return verificarCodigo(sheet, body);

    // Panel de colegio (panel-colegio/index.html) — su propio secreto por
    // colegio (PANEL_COLEGIO_SECRETS), validado adentro de resumenColegio.
    if (body.action === "resumen_colegio") return resumenColegio(sheet, body);

    // Resto de acciones — vienen del Worker, usan SHARED_SECRET.
    if (!body.secret || body.secret !== SHARED_SECRET) {
      return respond({ ok: false, error: "secret inválido" }, 401);
    }

    if (body.action === "guardar_borrador") return guardarBorrador(sheet, body);
    // NUEVO (28 oct 2026, pedido de Leonardo: regenerar casos reales con el
    // prompt ya corregido tras la revisión con Peter) — ver
    // regenerarBorrador() más abajo, es guardarBorrador() pero por idPlan en
    // vez de ts, y a propósito SIN el bloqueo de idempotencia: ese bloqueo
    // existe para no pisar una aprobación por una llamada automática
    // duplicada, pero acá SÍ queremos pisar el contenido — es el pedido
    // explícito. La llama el Worker, nunca equipo/ directo (el Worker valida
    // el panelSecret contra detalle_plan primero — ver /panel/regenerar).
    if (body.action === "regenerar_borrador") return regenerarBorrador(sheet, body);
    // NUEVO (28 oct 2026, pedido de Leonardo — "huella de regeneración"): el
    // Worker llama esto ANTES de encolar la regeneración real (marca
    // "iniciado") y en su catch si algo falla (marca "error: ..."). El
    // éxito lo marca regenerarBorrador() mismo, como parte de su guardado.
    if (body.action === "marcar_regeneracion") return marcarRegeneracion(sheet, body);
    if (body.action === "notificar_escalamiento") return notificarEscalamiento(sheet, body);
    // NUEVO (27 oct 2026, pedido de Leonardo): guarda el chat completo de
    // seguimiento — ver guardarMensajesChat() y la nota junto a ella.
    if (body.action === "guardar_mensaje_chat") return guardarMensajesChat(sheet, body);

    // Comportamiento original: agregar fila nueva capturada por Vita.
    if (!body.dx) {
      return respond({ ok: false, error: "dx faltante" }, 400);
    }
    var dx = body.dx;
    sheet.appendRow([
      body.ts || new Date().toISOString(),
      body.mode || "",
      dx.nombre_nino || "", dx.edad || "", dx.que_deseas_lograr || "",
      dx.preocupacion_principal || "", dx.origen || "", dx.sueno_y_movimiento || "",
      dx.senales_observables || "", dx.diagnosticos_y_atencion_medica || "",
      dx.cronologia_del_dia || "", dx.finde_y_gustos || "", dx.mente_y_pantallas || "",
      dx.emociones || "",
      "pendiente", "",       // estado, notas_peter
      "", "", "", "",        // id_plan, plan_json, decision, borrador_texto
      dx.nombre_padre || "", dx.email_padre || "", "",   // correo_enviado_ts vacío
      dx.consentimiento_ts || "",  // cuándo aceptó la pantalla de consentimiento (9 ago 2026)
      "", "",   // codigo_colegio, verificacion_ts — vacíos hasta que el padre verifique (ver verificarCodigo)
      // NUEVO (19 ago 2026): casilla SEPARADA de investigación — vacío si no
      // la marcó. OJO al agregar más columnas después de esta: hay que
      // mantener el mismo orden posicional que COLS, appendRow no usa
      // nombres de columna.
      dx.consentimiento_investigacion_ts || "",
      // NUEVO (2 oct 2026): snapshot de los hints del formulario en el
      // momento de esta captura (ver COLS.campos_snapshot). Vacío si el
      // Worker no lo mandó (ej. producto="srb3", o una versión vieja del
      // frontend que todavía no lo envía).
      body.camposSnapshot ? JSON.stringify(body.camposSnapshot) : ""
    ]);
    var lastRow = sheet.getLastRow();
    SpreadsheetApp.flush();
    return respond({ ok: true, lastRow: lastRow });
  } catch (err) {
    return respond({ ok: false, error: String(err) }, 500);
  }
}

// Envuelve las acciones del panel con el chequeo de PANEL_SECRET, para no
// repetirlo en cada función. "unauthorized:true" (no el texto exacto de
// "error") es lo que el Worker usa para decidir el status HTTP real —
// Apps Script/ContentService no puede mandar un status distinto de 200 (ver
// respond() más abajo), así que el Worker necesita ALGO en el cuerpo para
// distinguir "credencial inválida" de cualquier otro error. Un campo
// estructurado, no un string mágico: cualquier acción nueva que valide un
// secreto puede reusar el mismo patrón sin que el Worker necesite conocer
// el texto exacto de cada mensaje (ver handlePanelAction en vita-proxy.js).
function conPanelSecret(body, fn) {
  if (!body.panelSecret || !pinEsValido(body.panelSecret)) {
    return respond({ ok: false, error: "panelSecret inválido", unauthorized: true }, 401);
  }
  return fn();
}

// PIN personal de alguien en la pestaña "Equipo" (28 oct 2026, login por
// persona — ver loginEquipo() y la nota junto a "login_equipo" en doPost).
// Columna E (índice 4) de "Equipo" — Nombre/Rol/Correo/Notas/PIN. Se llena a
// mano en Sheets, igual que el resto de esa pestaña: no hay acción de panel
// para asignar PINes, es una lista de referencia simple.
function pinEsValido(pin) {
  if (pin === PANEL_SECRET) return true;
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Equipo");
  if (!hoja) return false;
  var data = hoja.getDataRange().getValues();
  for (var r = 1; r < data.length; r++) {
    if (data[r][4] && String(data[r][4]) === String(pin)) return true;
  }
  return false;
}

// Identifica DE QUIÉN es un PIN — para que equipo/index.html pueda saludar
// por nombre en vez de mostrar un "Panel Equipo" genérico. NO expone PINes
// (listarEquipo() tampoco los devuelve) — solo dice, para el PIN que
// mandaron, a quién pertenece. El PIN maestro no tiene nombre asociado: se
// acepta igual (fallback, nadie queda bloqueado) pero sin saludo personal.
function loginEquipo(body) {
  var pin = body.pin;
  if (!pin) return respond({ ok: false, error: "PIN incorrecto" }, 401);
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Equipo");
  if (hoja) {
    var data = hoja.getDataRange().getValues();
    for (var r = 1; r < data.length; r++) {
      if (data[r][4] && String(data[r][4]) === String(pin)) {
        return respond({ ok: true, nombre: data[r][0] || "", rol: data[r][1] || "" });
      }
    }
  }
  if (pin === PANEL_SECRET) return respond({ ok: true, nombre: "", rol: "" });
  return respond({ ok: false, error: "PIN incorrecto" }, 401);
}

// Lista los casos con borrador listo, esperando revisión — solo lo mínimo
// para elegir cuál abrir, no expone la fila completa.
function listarPendientes(sheet) {
  var data = sheet.getDataRange().getValues();
  var out = [];
  for (var r = 1; r < data.length; r++) {
    if (data[r][COLS.decision] === "pendiente_aprobacion") {
      out.push({
        idPlan: data[r][COLS.id_plan],
        nombreNino: data[r][COLS.nombre_nino],
        edad: data[r][COLS.edad],
        nombrePadre: data[r][COLS.nombre_padre],
        preocupacionPrincipal: data[r][COLS.preocupacion_principal],
        timestamp: data[r][COLS.timestamp],
        verificacionTs: data[r][COLS.verificacion_ts],
      });
    }
  }
  out.sort(function (a, b) { return new Date(b.timestamp) - new Date(a.timestamp); }); // más recientes primero
  return respond({ ok: true, pendientes: out });
}

function encontrarFilaPorIdPlan(sheet, idPlan) {
  var data = sheet.getDataRange().getValues();
  for (var r = 1; r < data.length; r++) {
    if (String(data[r][COLS.id_plan]) === String(idPlan)) return { row: r + 1, values: data[r] };
  }
  return null;
}

// Guarda ediciones (hábitos incluidos/excluidos, texto del mensaje) SIN
// aprobar — para que Peter pueda ajustar y volver más tarde sin apurarse.
function guardarEdicionPlan(sheet, body) {
  var idPlan = body.idPlan;
  if (!idPlan) return respond({ ok: false, error: "idPlan faltante" }, 400);
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return respond({ ok: false, error: "no encontrado" }, 404);
  var row = found.row;
  if (body.plan) sheet.getRange(row, COLS.plan_json + 1).setValue(JSON.stringify(body.plan));

  SpreadsheetApp.flush();
  return respond({ ok: true });
}

// Aprueba: guarda los cambios finales (si vienen) + decision=aprobado +
// dispara el correo real — mismo camino que el trigger de edición manual,
// para que ambas formas de aprobar den exactamente el mismo resultado.
// BLOQUEO DE ELEGIBILIDAD (10 ago 2026): no se puede aprobar un caso cuya
// familia todavía no verificó el código de su colegio — así se garantiza
// que la recomendación se retiene mientras no se confirme que tiene
// derecho a recibirla, sin depender de que alguien se acuerde de revisarlo
// a ojo en el panel.
function aprobarPlan(sheet, body) {
  var idPlan = body.idPlan;
  if (!idPlan) return respond({ ok: false, error: "idPlan faltante" }, 400);
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return respond({ ok: false, error: "no encontrado" }, 404);
  if (!MODO_PRUEBAS && !found.values[COLS.verificacion_ts]) {
    return respond({ ok: false, error: "no se puede aprobar: el padre todavía no verificó el código de su colegio" }, 400);
  }
  var row = found.row;
  if (body.plan) sheet.getRange(row, COLS.plan_json + 1).setValue(JSON.stringify(body.plan));
  if (typeof body.mensaje === "string") sheet.getRange(row, COLS.borrador_texto + 1).setValue(body.mensaje);
  sheet.getRange(row, COLS.decision + 1).setValue("aprobado");
  SpreadsheetApp.flush();
  var resultado = enviarPlanAprobado(sheet, row);
  return respond({ ok: true, envio: resultado });
}

// Descarta un caso (borra la fila) — para casos de prueba, duplicados, o
// algo que claramente no aplica. Bloqueado a propósito si a ESE caso YA se
// le mandó el correo real a una familia (correo_enviado_ts tiene valor): no
// se puede borrar un plan que una familia real ya puede estar viendo, ni
// por accidente ni por confusión de idPlan. OJO: el bloqueo es por
// correo_enviado_ts, NO por decision="aprobado" — un caso puede quedar
// "aprobado" sin que el correo haya salido de verdad (ej. sin correo del
// padre capturado, o un caso de prueba aprobado a propósito para verificar
// el flujo) y ESE sí se puede descartar.
function descartarPendiente(sheet, body) {
  var idPlan = body.idPlan;
  if (!idPlan) return respond({ ok: false, error: "idPlan faltante" }, 400);
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return respond({ ok: false, error: "no encontrado" }, 404);
  if (found.values[COLS.correo_enviado_ts]) {
    return respond({ ok: false, error: "no se puede descartar: ya se le envió el correo a la familia" }, 400);
  }
  sheet.deleteRow(found.row);
  return respond({ ok: true });
}

function guardarBorrador(sheet, body) {
  var ts = body.ts;
  if (!ts) return respond({ ok: false, error: "ts faltante" }, 400);

  var data = sheet.getDataRange().getValues();
  for (var r = 1; r < data.length; r++) {
    if (String(data[r][COLS.timestamp]) === String(ts)) {
      var rowNum = r + 1;

      // IDEMPOTENCIA (14 ago 2026): si esta fila YA tiene un id_plan, no
      // generar uno nuevo ni pisar nada — se devuelve el que ya existe. Sin
      // esto, dos llamadas para el mismo ts (la generación automática en
      // segundo plano de /log + una manual, o cualquier reintento futuro)
      // generan un idPlan NUEVO cada vez y resetean decision a
      // "pendiente_aprobacion" — lo que borra silenciosamente una
      // aprobación ya hecha, e invalida cualquier link ya compartido con
      // una familia. Encontrado en vivo el 14 ago 2026 validando el panel
      // de colegio.
      var idPlanExistente = data[r][COLS.id_plan];
      if (idPlanExistente) {
        return respond({ ok: true, idPlan: idPlanExistente, row: rowNum, yaExistia: true });
      }

      var idPlan = Utilities.getUuid().replace(/-/g, "").slice(0, 16);
      sheet.getRange(rowNum, COLS.id_plan + 1).setValue(idPlan);
      sheet.getRange(rowNum, COLS.plan_json + 1).setValue(JSON.stringify(body.plan || {}));
      sheet.getRange(rowNum, COLS.decision + 1).setValue("pendiente_aprobacion");
      sheet.getRange(rowNum, COLS.borrador_texto + 1).setValue(body.mensaje || "");
      sheet.getRange(rowNum, COLS.estado + 1).setValue("borrador_listo");
      SpreadsheetApp.flush();
      enviarSolicitudVerificacion(sheet, rowNum, idPlan);
      return respond({ ok: true, idPlan: idPlan, row: rowNum });
    }
  }
  return respond({ ok: false, error: "no se encontró la fila con ese ts" }, 404);
}

// Sobrescribe el borrador de un caso que YA EXISTE (idPlan conocido) con un
// contenido recién regenerado — a diferencia de guardarBorrador() (que se
// niega a pisar un idPlan existente, a propósito, para que un reintento
// automático nunca borre una aprobación), esta acción SÍ pisa el contenido:
// es un pedido explícito del equipo, no un reintento accidental. SIEMPRE
// vuelve decision a "pendiente_aprobacion", incluso si el caso ya estaba
// "aprobado" — el contenido nuevo tiene que pasar otra vez por la revisión
// de Peter antes de llegar a la familia, igual que cualquier borrador nuevo
// (regla dura del proyecto: HITL obligatorio). No toca correo_enviado_ts ni
// verificacion_ts — son historial de lo que ya pasó, no se borran.
function regenerarBorrador(sheet, body) {
  var idPlan = body.idPlan;
  if (!idPlan) return respond({ ok: false, error: "idPlan faltante" }, 400);
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return respond({ ok: false, error: "no encontrado" }, 404);
  var row = found.row;
  sheet.getRange(row, COLS.plan_json + 1).setValue(JSON.stringify(body.plan || {}));
  sheet.getRange(row, COLS.decision + 1).setValue("pendiente_aprobacion");
  sheet.getRange(row, COLS.borrador_texto + 1).setValue(body.mensaje || "");
  sheet.getRange(row, COLS.estado + 1).setValue("borrador_listo");
  sheet.getRange(row, COLS.regeneracion_estado + 1).setValue("ok " + new Date().toISOString());
  SpreadsheetApp.flush();
  return respond({ ok: true, idPlan: idPlan, row: row });
}

// rastro que no avisar nada.
function notificarEscalamiento(sheet, body) {
  try {
    MailApp.sendEmail({
      to: "leonardo@ledelab.group",
      subject: "🚨 Vita escaló algo — " + (body.motivo || "sin especificar") + (body.nombreNino ? " — " + body.nombreNino : ""),
      body:
        "Motivo: " + (body.motivo || "sin especificar") + "\n" +
        "Niño/a: " + (body.nombreNino || "(sin dato)") + "\n" +
        "id_plan: " + (body.idPlan || "(sin dato)") + "\n\n" +
        "Mensaje del padre/madre:\n" + (body.mensajePadre || "(sin mensaje)"),
    });
    if (body.idPlan) {
      try { registrarNovedad(sheet, body.idPlan, body.motivo, body.mensajePadre); } catch (exNovedad) {}
    }
    return respond({ ok: true });
  } catch (err) {
    return respond({ ok: false, error: String(err) }, 500);
  }
}

// Guarda uno o más mensajes del chat de seguimiento (27 oct 2026, pedido de
// Leonardo — "el chat completo también", para que Vita recuerde la
// conversación entre sesiones en la pestaña Vita de app/index.html, y para
// mostrarla si hace falta en Progreso). Mismo patrón append-only que
// registrarNovedad() de abajo, en su propia columna (chat_json) — nunca se
// sobreescribe. SHARED_SECRET, el mismo nivel que notificar_escalamiento:
// NO es una acción de panel (nunca lleva panelSecret), la dispara la app
// pública de la familia a través del Worker, después de cada turno completo
// (nunca mensaje por mensaje, para no multiplicar llamadas).
function guardarMensajesChat(sheet, body) {
  var idPlan = body.idPlan;
  var mensajes = body.mensajes;
  if (!idPlan) return respond({ ok: false, error: "idPlan faltante" }, 400);
  if (!Array.isArray(mensajes) || mensajes.length === 0) return respond({ ok: false, error: "mensajes faltante" }, 400);
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return respond({ ok: false, error: "no encontrado" }, 404);
  var actuales = [];
  try { actuales = JSON.parse(found.values[COLS.chat_json] || "[]"); } catch (ex) {}
  mensajes.forEach(function (m) {
    actuales.push({ ts: m.ts || new Date().toISOString(), role: m.role || "", content: m.content || "" });
  });
  sheet.getRange(found.row, COLS.chat_json + 1).setValue(JSON.stringify(actuales));
  SpreadsheetApp.flush();
  return respond({ ok: true });
}

// Agrega una entrada al historial de novedades del caso (nunca sobreescribe
// las anteriores). Ver detallePlan() para cómo el panel las lee de vuelta.
function registrarNovedad(sheet, idPlan, motivo, resumen) {
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return;
  var actuales = [];
  try { actuales = JSON.parse(found.values[COLS.novedades_json] || "[]"); } catch (ex) {}
  actuales.push({ ts: new Date().toISOString(), motivo: motivo || "sin especificar", resumen: resumen || "" });
  sheet.getRange(found.row, COLS.novedades_json + 1).setValue(JSON.stringify(actuales));
  SpreadsheetApp.flush();
}

// NUEVO (27 oct 2026, pedido de Leonardo): borra UNA novedad del caso, por
// posición — para descartar una entrada de prueba o una ya atendida, sin
// tener que editar el Sheet a mano. Es justo el tipo de edición directa que
// el panel existe para evitar (ver la nota grande al inicio del archivo) —
// de ahí que valga la pena esta acción en vez de pedirle a alguien que
// abra la hoja de cálculo y borre una celda.
function eliminarNovedad(sheet, body) {
  var idPlan = body.idPlan;
  var index = body.index;
  if (!idPlan) return respond({ ok: false, error: "idPlan faltante" }, 400);
  if (typeof index !== "number" || index < 0) return respond({ ok: false, error: "index faltante o inválido" }, 400);
  var found = encontrarFilaPorIdPlan(sheet, idPlan);
  if (!found) return respond({ ok: false, error: "no encontrado" }, 404);
  var actuales = [];
  try { actuales = JSON.parse(found.values[COLS.novedades_json] || "[]"); } catch (ex) {}
  if (index >= actuales.length) return respond({ ok: false, error: "esa novedad ya no existe" }, 404);
  actuales.splice(index, 1);
  sheet.getRange(found.row, COLS.novedades_json + 1).setValue(JSON.stringify(actuales));
  SpreadsheetApp.flush();
  return respond({ ok: true, novedades: actuales });
}

// Lista del equipo (27 oct 2026, pedido de Leonardo — sección "Equipo" de
// equipo/index.html) — referencia de SOLO LECTURA de quién tiene acceso
// hoy. El login sigue siendo un único PANEL_SECRET compartido; esto no lo
// cambia, solo lo hace visible. Vive en su propia pestaña "Equipo" del
// mismo Sheet (no la de casos) — se crea sola con encabezados la primera
// vez que se llama, así no hace falta preparar nada a mano antes de
// desplegar. Agregar o quitar personas se hace escribiendo directo en esa
// pestaña de Sheets — a diferencia de los casos clínicos, esta es una
// lista de referencia simple sin lógica que se pueda romper por un valor
// mal puesto, así que no hace falta una acción de panel para editarla.
function listarEquipo() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = ss.getSheetByName("Equipo");
  if (!hoja) {
    hoja = ss.insertSheet("Equipo");
    hoja.appendRow(["Nombre", "Rol", "Correo", "Notas"]);
    hoja.setFrozenRows(1);
    SpreadsheetApp.flush();
  }
  var data = hoja.getDataRange().getValues();
  var out = [];
  for (var r = 1; r < data.length; r++) {
    if (!data[r][0]) continue; // fila vacía (ej. separador a mano) — se ignora
    out.push({
      nombre: data[r][0] || "",
      rol: data[r][1] || "",
      correo: data[r][2] || "",
      notas: data[r][3] || "",
    });
  }
  return respond({ ok: true, equipo: out });
}

function doGet(e) {
  try {
    var id = e.parameter.id;
    if (!id) return respond({ ok: false, error: "id faltante" }, 400);

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    var data = sheet.getDataRange().getValues();

    for (var r = 1; r < data.length; r++) {
      if (String(data[r][COLS.id_plan]) === String(id)) {
        var decision = data[r][COLS.decision];
        if (decision !== "aprobado") {
          return respond({ ok: false, error: "aún no aprobado" }, 403);
        }
        var plan = {};
        try { plan = JSON.parse(data[r][COLS.plan_json] || "{}"); } catch (ex) {}
        // El padre NUNCA ve los hábitos que Peter excluyó en el panel — se
        // guardan todos (para que Peter pueda reconsiderar antes de
        // aprobar), pero acá se filtran. Sin "incluido" (casos viejos, antes
        // de que existiera el panel) se trata como incluido.
        if (plan.habitos) {
          plan.habitos = plan.habitos.filter(function (h) { return h.incluido !== false; });
        }
        // NUEVO (27 oct 2026, pedido de Leonardo — pestaña Progreso y
        // continuidad de la pestaña Vita en app/index.html): el padre ya
        // podía ver estos datos de otras formas (las novedades ya se le
        // mandaban por correo; el chat ya lo escribió él mismo) — acá solo
        // se le devuelven de vuelta, no es información nueva expuesta.
        var novedades = [];
        try { novedades = JSON.parse(data[r][COLS.novedades_json] || "[]"); } catch (ex) {}
        var chatHistory = [];
        try { chatHistory = JSON.parse(data[r][COLS.chat_json] || "[]"); } catch (ex) {}
        // NUEVO (28 oct 2026, pedido de Leonardo — Progreso v1, línea de
        // etapas): fechas que ya existían en el Sheet, nunca expuestas antes
        // en /plan — el padre ya las conoce (cuándo llenó el formulario,
        // cuándo le llegó el correo de aprobación), no es dato nuevo.
        return respond({
          ok: true,
          nombreNino: data[r][COLS.nombre_nino],
          mensaje: data[r][COLS.borrador_texto],
          plan: plan,
          novedades: novedades,
          chatHistory: chatHistory,
          timestamp: data[r][COLS.timestamp] || "",
          correoEnviadoTs: data[r][COLS.correo_enviado_ts] || "",

    return respond({ ok: false, error: "no encontrado" }, 404);
  } catch (err) {
    return respond({ ok: false, error: String(err) }, 500);
  }
}

    });

  // (MailApp manda texto plano, no renderiza nada). Quita los ** de negrita,
  // los * de énfasis y las líneas que son solo guiones o signos de igual.
  function aTextoPlano(t) {
    return String(t || "")
      .replace(/\*\*(.+?)\*\*/g, "$1")
      .replace(/(^|\s)\*(\S[^*]*?)\*(?=\s|$|[.,;:])/g, "$1$2")
      .split("\n")
      .filter(function (linea) { return !/^\s*[-=_─═]{3,}\s*$/.test(linea); })
      .join("\n")
      .replace(/\n{3,}/g, "\n\n")
      .trim();

    }).join("\n\n");

  MailApp.sendEmail({
    to: emailPadre,
    name: "Bienestar 360",

    body:

      "Ya está listo el plan de " + (nombreNino || "tu hijo/a") + ", revisado y aprobado por Peter Álvarez.\n\n" +

  // forma de ver qué le llegó de verdad al padre/madre después de su

// A propósito SÍ manda un correo de resumen al equipo cuando purga algo —

// NO corre sola con pegar el código — igual que onAprobado, es un trigger

//   1. En el editor de Apps Script, ícono del reloj (⏰, "Activadores").
//   2. + Agregar activador.
//   3. Función: purgarInactivosProgramado · Fuente del evento: Basado en
//      tiempo · Tipo de activador basado en tiempo: Temporizador mensual
//      (elige el día del mes y la hora) · Guardar.
//   4. Autoriza permisos si te lo pide (es tu propio script).
// Mientras no instales el activador, esta función existe en el código pero
// nunca se ejecuta sola — no hay riesgo de que borre nada sin que lo
// actives a propósito.
// ============================================================
var MESES_INACTIVIDAD_PURGA = 24;

function purgarInactivos(sheet) {
  var data = sheet.getDataRange().getValues();
  var limite = new Date();
  limite.setMonth(limite.getMonth() - MESES_INACTIVIDAD_PURGA);
  var purgados = [];
  // De abajo hacia arriba: borrar una fila corre hacia arriba las de abajo,
  // así que iterar en reversa evita saltarse filas por el corrimiento.
  for (var r = data.length - 1; r >= 1; r--) {
    if (data[r][COLS.decision] !== "aprobado") continue;
    var fechaRef = data[r][COLS.correo_enviado_ts] || data[r][COLS.timestamp];
    if (!fechaRef) continue;
    var fecha = new Date(fechaRef);
    if (isNaN(fecha.getTime())) continue;
    if (fecha < limite) {
      purgados.push({ nombreNino: data[r][COLS.nombre_nino] || "(sin nombre)", fechaRef: fechaRef });
      sheet.deleteRow(r + 1);
    }
  }
  if (purgados.length > 0) {
    MailApp.sendEmail({
      to: "leonardo@ledelab.group",
      subject: "🗑️ Purga automática por inactividad — " + purgados.length + " caso(s)",
      body: "Se borraron " + purgados.length + " caso(s) por pasar " + MESES_INACTIVIDAD_PURGA +
        " meses sin actividad conocida (política de retención aprobada 9 ago 2026):\n\n" +
        purgados.map(function (p) { return "- " + p.nombreNino + " (última actividad: " + p.fechaRef + ")"; }).join("\n"),
    });
  }
  return purgados;
}

// Función objetivo del trigger de tiempo — sin argumentos, porque los

function purgarInactivosProgramado() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  purgarInactivos(sheet);
}

    return respond({ ok: false, error: "código inválido — verifica con tu colegio" }, 400);
  }

  sheet.getRange(found.row, COLS.verificacion_ts + 1).setValue(new Date().toISOString());
  SpreadsheetApp.flush();
  return respond({ ok: true });
}

// PANEL DE COLEGIO (14 ago 2026, pedido de Leonardo) — para el rector o la
// persona delegada del colegio, DISTINTO del panel de Peter. Muestra SOLO
// agregados del colegio, nunca el caso de un niño/a individual — es la
// promesa de privacidad ya hecha en las propuestas comerciales (Royal
// School, FUMAVI, Altamira, Medalla Milagrosa), y además protege contra un
// riesgo real que identificamos: en varios colegios el rector o delegado es
// también padre/madre de una familia inscrita, así que conoce personalmente
// a otras familias del piloto.
//
// UMBRAL DE PRIVACIDAD (n>=5, UMBRAL_PRIVACIDAD_COLEGIO): con cohortes
// chicas, cualquier desagregado (aprobados/en proceso, edades, evolución
// mensual) puede volverse identificable para alguien que conoce a las
// familias — un agregado de "1 de 3" no es anónimo para quien conoce a las
// otras 2. Por debajo del umbral SOLO se devuelve el total (un número de
// inscripción no es sensible por sí solo) — nada desagregado. Es un umbral
// de PRIVACIDAD (evita reidentificación por eliminación), NO de
// representatividad estadística — no confundir los dos conceptos al
// explicarle esto a un colegio.
//
// Solo cuenta familias VERIFICADAS de ese colegio (verificacion_ts con
// valor) — un borrador sin confirmar todavía podría terminar siendo de otro
// colegio, o nunca completarse.
//
// LO QUE NO HACE A PROPÓSITO: no desagrega por "señales observadas" ni
// ningún otro campo de texto libre del formulario — esos campos los llena
// cada familia con sus propias palabras (no son checkboxes), así que
// cualquier conteo por palabra clave sería una afirmación poco confiable
// sobre datos de salud. Mejor no mostrar ese dato que mostrar uno impreciso.
// ============================================================
var UMBRAL_PRIVACIDAD_COLEGIO = 5;

function resumenColegio(sheet, body) {
  var codigo = String(body.codigo || "").trim();
  var pin = String(body.panelColegioSecret || "");
  if (!codigo) return respond({ ok: false, error: "código de colegio faltante" }, 400);
  if (!PANEL_COLEGIO_SECRETS[codigo] || PANEL_COLEGIO_SECRETS[codigo] !== pin) {
    // unauthorized:true, no el texto exacto — ver la nota junto a conPanelSecret.
    return respond({ ok: false, error: "panelColegioSecret inválido", unauthorized: true }, 401);
  }

  var data = sheet.getDataRange().getValues();
  var filas = [];
  for (var r = 1; r < data.length; r++) {
    if (data[r][COLS.codigo_colegio] === codigo && data[r][COLS.verificacion_ts]) {
      filas.push(data[r]);
    }
  }

  var total = filas.length;
  var out = { ok: true, colegio: codigo, total: total, umbral: UMBRAL_PRIVACIDAD_COLEGIO };

  if (total < UMBRAL_PRIVACIDAD_COLEGIO) {
    out.muestraInsuficiente = true;
    return respond(out);
  }

  var aprobados = 0;
  var porEdad = { "2-5": 0, "6-9": 0, "10-13": 0, "14-17": 0, "18-20": 0, sin_dato: 0 };
  var porMes = {}; // "2026-08" -> conteo, por fecha de verificación

  filas.forEach(function (v) {
    if (v[COLS.decision] === "aprobado") aprobados++;

    var edadNum = parseInt(String(v[COLS.edad]).replace(/[^0-9]/g, ""), 10);
    if (isNaN(edadNum)) porEdad.sin_dato++;
    else if (edadNum <= 5) porEdad["2-5"]++;
    else if (edadNum <= 9) porEdad["6-9"]++;
    else if (edadNum <= 13) porEdad["10-13"]++;
    else if (edadNum <= 17) porEdad["14-17"]++;
    else porEdad["18-20"]++;

    var fecha = new Date(v[COLS.verificacion_ts]);
    if (!isNaN(fecha.getTime())) {
      var mes = fecha.getFullYear() + "-" + ("0" + (fecha.getMonth() + 1)).slice(-2);
      porMes[mes] = (porMes[mes] || 0) + 1;
    }
  });

  out.aprobados = aprobados;
  out.enProceso = total - aprobados;
  out.porEdad = porEdad;
  out.evolucionMensual = porMes;
  return respond(out);
}
